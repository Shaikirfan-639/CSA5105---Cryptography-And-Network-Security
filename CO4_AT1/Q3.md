'''Q3. IPSec Security Association (SA) Management'''

print("========================================")
print(" Q3 - IPSEC SECURITY ASSOCIATION")
print("========================================")

# Security Association Database
SA_DATABASE = {}


def create_sa(sa_id, source, destination, protocol, encryption):
    if sa_id in SA_DATABASE:
        print("\nSA already exists.")
        return

    SA_DATABASE[sa_id] = {
        "source": source,
        "destination": destination,
        "protocol": protocol,
        "encryption": encryption
    }

    print(f"\nSecurity Association {sa_id} created successfully.")


def lookup_sa(sa_id):
    if sa_id in SA_DATABASE:
        print(f"\nSA {sa_id} found:")
        for key, value in SA_DATABASE[sa_id].items():
            print(f"{key}: {value}")
    else:
        print(f"\nSA {sa_id} not found.")


def delete_sa(sa_id):
    if sa_id in SA_DATABASE:
        del SA_DATABASE[sa_id]
        print(f"\nSecurity Association {sa_id} deleted.")
    else:
        print(f"\nSA {sa_id} does not exist.")


def display_all_sa():
    print("\nCurrent Security Association Database:")

    if not SA_DATABASE:
        print("No active Security Associations.")
        return

    for sa_id, details in SA_DATABASE.items():
        print(f"\nSA ID: {sa_id}")

        for key, value in details.items():
            print(f"  {key}: {value}")


# --------------------------------
# Demonstration
# --------------------------------

create_sa(
    1001,
    "192.168.1.10",
    "192.168.1.20",
    "ESP",
    "AES-256"
)

create_sa(
    1002,
    "10.0.0.10",
    "10.0.0.20",
    "AH",
    "HMAC-SHA256"
)

display_all_sa()

# Lookup
lookup_sa(1001)

# Delete
delete_sa(1002)

# Display after deletion
display_all_sa()
