'''Q5. Email Spam Classification'''

import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, classification_report

print("========================================")
print(" Q5 - EMAIL SPAM CLASSIFICATION")
print("========================================")

# ---------------------------------------
# Synthetic email dataset
# ---------------------------------------

emails = [
    "Subject: Win a free iPhone now! Click here to claim your prize",
    "Subject: Congratulations! You have won $10000. Claim now",
    "Subject: Urgent! You have won a lottery prize",
    "Subject: Free gift waiting for you. Click now",
    "Subject: Earn money quickly from home",
    "Subject: Exclusive offer! Claim your free reward",
    "Subject: You are selected for a cash prize",
    "Subject: Limited time offer. Buy now and save money",

    "Subject: Meeting scheduled for tomorrow at 10 AM",
    "Subject: Your assignment submission deadline is Friday",
    "Subject: Project discussion with the development team",
    "Subject: Monthly bank account statement",
    "Subject: College examination timetable",
    "Subject: Team meeting minutes and action items",
    "Subject: Your appointment is confirmed",
    "Subject: Please find the project report attached"
]

labels = [
    "spam",
    "spam",
    "spam",
    "spam",
    "spam",
    "spam",
    "spam",
    "spam",

    "legitimate",
    "legitimate",
    "legitimate",
    "legitimate",
    "legitimate",
    "legitimate",
    "legitimate",
    "legitimate"
]

# Create DataFrame
data = pd.DataFrame({
    "email": emails,
    "label": labels
})

print("\nDataset:")
print(data)

# ---------------------------------------
# Convert text into numerical features
# ---------------------------------------

vectorizer = TfidfVectorizer(
    lowercase=True,
    stop_words="english"
)

X = vectorizer.fit_transform(data["email"])
y = data["label"]

# ---------------------------------------
# Train/Test Split
# ---------------------------------------

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.25,
    random_state=42,
    stratify=y
)

# ---------------------------------------
# Train classifier
# ---------------------------------------

model = MultinomialNB()
model.fit(X_train, y_train)

# ---------------------------------------
# Prediction
# ---------------------------------------

predictions = model.predict(X_test)

accuracy = accuracy_score(y_test, predictions)

print("\nClassification Accuracy:")
print(f"{accuracy * 100:.2f}%")

print("\nClassification Report:")
print(classification_report(y_test, predictions))

# ---------------------------------------
# Test new emails
# ---------------------------------------

new_emails = [
    "Congratulations! You won a free cash prize. Click now",
    "Please attend the project meeting tomorrow morning",
    "Urgent claim your free reward immediately",
    "Your college assignment has been submitted successfully"
]

new_features = vectorizer.transform(new_emails)
new_predictions = model.predict(new_features)

print("\nNew Email Classification:")

for email, result in zip(new_emails, new_predictions):
