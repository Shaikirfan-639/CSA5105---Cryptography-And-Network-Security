'''Q2. IPSec ESP — Encryption and Decryption'''

from cryptography.hazmat.primitives.ciphers.aead import AESGCM
import os

print("========================================")
print(" Q2 - IPSEC ESP SIMULATION")
print("========================================")

# Generate AES-256 key
key = AESGCM.generate_key(bit_length=256)

# Create AES-GCM cipher
aes = AESGCM(key)

# Random nonce
nonce = os.urandom(12)

# IP packet payload
payload = b"Confidential IP Packet Payload"

print("\nOriginal Payload:")
print(payload.decode())

# ESP Encryption
encrypted_payload = aes.encrypt(
    nonce,
    payload,
    None
)

print("\nEncrypted ESP Payload:")
print(encrypted_payload.hex())

# ESP Decryption
decrypted_payload = aes.decrypt(
    nonce,
    encrypted_payload,
    None
)

print("\nDecrypted Payload:")
print(decrypted_payload.decode())

# Verification
if decrypted_payload == payload:
    print("\nESP Result: Encryption and decryption successful.")
else:
    print("\nESP Result: Verification failed.")
