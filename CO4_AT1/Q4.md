'''Q4. Simplified PGP Secure Email System'''

from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives import serialization
import os

print("========================================")
print(" Q4 - SIMPLIFIED PGP EMAIL SYSTEM")
print("========================================")

# ---------------------------------------
# 1. Generate RSA key pairs
# ---------------------------------------

sender_private_key = rsa.generate_private_key(
    public_exponent=65537,
    key_size=2048
)

sender_public_key = sender_private_key.public_key()

receiver_private_key = rsa.generate_private_key(
    public_exponent=65537,
    key_size=2048
)

receiver_public_key = receiver_private_key.public_key()

print("\nRSA key pairs generated.")

# ---------------------------------------
# 2. Email message
# ---------------------------------------

message = b"Hello Receiver, this is a secure PGP email."

print("\nOriginal Email:")
print(message.decode())

# ---------------------------------------
# 3. Digital Signature
# ---------------------------------------

signature = sender_private_key.sign(
    message,
    padding.PSS(
        mgf=padding.MGF1(hashes.SHA256()),
        salt_length=padding.PSS.MAX_LENGTH
    ),
    hashes.SHA256()
)

print("\nDigital signature generated.")

# ---------------------------------------
# 4. AES encryption of email
# ---------------------------------------

aes_key = AESGCM.generate_key(bit_length=256)
aes = AESGCM(aes_key)

nonce = os.urandom(12)

encrypted_message = aes.encrypt(
    nonce,
    message,
    None
)

print("Email encrypted using AES.")

# ---------------------------------------
# 5. RSA encryption of AES key
# ---------------------------------------

encrypted_aes_key = receiver_public_key.encrypt(
    aes_key,
    padding.OAEP(
        mgf=padding.MGF1(algorithm=hashes.SHA256()),
        algorithm=hashes.SHA256(),
        label=None
    )
)

print("AES key encrypted using receiver's RSA public key.")

# ---------------------------------------
# RECEIVER SIDE
# ---------------------------------------

# 6. Decrypt AES key using RSA private key
decrypted_aes_key = receiver_private_key.decrypt(
    encrypted_aes_key,
    padding.OAEP(
        mgf=padding.MGF1(algorithm=hashes.SHA256()),
        algorithm=hashes.SHA256(),
        label=None
    )
)

# 7. Decrypt email
receiver_aes = AESGCM(decrypted_aes_key)

decrypted_message = receiver_aes.decrypt(
    nonce,
    encrypted_message,
    None
)

print("\nDecrypted Email:")
print(decrypted_message.decode())

# ---------------------------------------
# 8. Verify Digital Signature
# ---------------------------------------

try:
    sender_public_key.verify(
        signature,
        decrypted_message,
        padding.PSS(
            mgf=padding.MGF1(hashes.SHA256()),
            salt_length=padding.PSS.MAX_LENGTH
        ),
        hashes.SHA256()
    )

    print("\nDigital Signature:")
    print("VALID - Sender authenticated and message integrity verified.")

except Exception:
    print("\nDigital Signature:")
    print("INVALID - Message may have been modified.")

print("\nPGP Simulation Completed Successfully.")
