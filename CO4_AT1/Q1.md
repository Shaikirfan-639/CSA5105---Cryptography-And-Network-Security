'''Q1. IPSec Authentication Header (AH) Simulation'''


import hmac
import hashlib

print("========================================")
print(" Q1 - IPSEC AUTHENTICATION HEADER (AH)")
print("========================================")

# Secret authentication key
secret_key = b"AH_SECRET_KEY_123"

# Simulated IP packet
ip_packet = "Source: 192.168.1.10 | Destination: 192.168.1.20 | Data: Hello Server"

print("\nOriginal IP Packet:")
print(ip_packet)

# Generate MAC
mac = hmac.new(
    secret_key,
    ip_packet.encode(),
    hashlib.sha256
).hexdigest()

print("\nGenerated AH MAC:")
print(mac)

# Verification function
def verify_ah(packet, received_mac):
    calculated_mac = hmac.new(
        secret_key,
        packet.encode(),
        hashlib.sha256
    ).hexdigest()

    return hmac.compare_digest(calculated_mac, received_mac)

# Verify original packet
result = verify_ah(ip_packet, mac)

print("\nVerification of original packet:")
print("AUTHENTIC - Packet integrity verified" if result
      else "INVALID - Packet has been modified")

# Simulate packet modification
modified_packet = "Source: 192.168.1.10 | Destination: 192.168.1.20 | Data: HACKED"

print("\nModified IP Packet:")
print(modified_packet)

result = verify_ah(modified_packet, mac)

print("\nVerification of modified packet:")
print("AUTHENTIC - Packet integrity verified" if result
      else "INVALID - Packet modification detected")
